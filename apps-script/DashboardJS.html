const WEB_APP_URL = ''; function showError(message){ const el = document.getElementById('errorBox'); el.textContent = message || 'Terjadi kesalahan.'; el.classList.remove('hidden'); } function redirectLogin(){ window.top.location.href = WEB_APP_URL + '?page=login'; } function loadCurrentUser(){ google.script.run .withSuccessHandler(function(user){ if (!user || user.loggedIn !== true){ redirectLogin(); return; } document.getElementById('welcome').textContent = 'Selamat datang, ' + user.nama; document.getElementById('roleText').textContent = 'Sesi aktif pada CMS BMT Al-Muhajirin.'; document.getElementById('userName').textContent = user.nama; document.getElementById('userEmail').textContent = user.email; document.getElementById('userRole').textContent = user.role; document.getElementById('roleBadge').textContent = user.role; if (user.role === 'admin'){ document.getElementById('roleText').textContent = 'Akses administrator aktif.'; } else { document.getElementById('roleText').textContent = 'Akses pengguna aktif sesuai role user.'; } }) .withFailureHandler(function(error){ showError(error && error.message ? error.message : 'Sesi tidak dapat diperiksa.'); setTimeout(redirectLogin, 1200); }) .getCurrentUser(); } function logout(){ document.getElementById('logoutBtn').disabled = true; google.script.run .withSuccessHandler(function(){ redirectLogin(); }) .withFailureHandler(function(){ redirectLogin(); }) .doLogout(); } loadCurrentUser();
